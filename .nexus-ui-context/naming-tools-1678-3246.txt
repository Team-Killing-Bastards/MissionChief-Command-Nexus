01679:     function getToolViewportBounds() {
01680:         const viewport = window.visualViewport;
01681:         const left = Math.max(0, Number(viewport?.offsetLeft || 0));
01682:         const top = Math.max(0, Number(viewport?.offsetTop || 0));
01683:         const width = Math.max(1, Number(viewport?.width || window.innerWidth || 1));
01684:         const height = Math.max(1, Number(viewport?.height || window.innerHeight || 1));
01685: 
01686:         return {
01687:             left,
01688:             top,
01689:             width,
01690:             height,
01691:             right: left + width,
01692:             bottom: top + height
01693:         };
01694:     }

01696:     function clampToolPanelToViewport(panel) {
01697:         if (!panel?.isConnected) return;
01698: 
01699:         // Before the first drag, iOS Safari is positioned entirely by the
01700:         // safe-area CSS below. Only clamp an explicitly dragged mobile panel.
01701:         if (
01702:             panel.classList.contains('mc-ios-safari')
01703:             && !panel.style.left
01704:         ) {
01705:             return;
01706:         }
01707: 
01708:         const bounds = getToolViewportBounds();
01709:         let rect = panel.getBoundingClientRect();
01710:         const maximumWidth = Math.max(260, bounds.width - 16);
01711: 
01712:         if (rect.width > maximumWidth) {
01713:             panel.style.width = `${maximumWidth}px`;
01714:             rect = panel.getBoundingClientRect();
01715:         }
01716: 
01717:         const maximumLeft = Math.max(
01718:             bounds.left,
01719:             bounds.right - rect.width
01720:         );
01721:         const maximumTop = Math.max(
01722:             bounds.top,
01723:             bounds.bottom - rect.height
01724:         );
01725:         const nextLeft = Math.min(
01726:             maximumLeft,
01727:             Math.max(bounds.left, rect.left)
01728:         );
01729:         const nextTop = Math.min(
01730:             maximumTop,
01731:             Math.max(bounds.top, rect.top)
01732:         );
01733: 
01734:         panel.style.left = `${nextLeft}px`;
01735:         panel.style.top = `${nextTop}px`;
01736:         panel.style.right = 'auto';
01737:     }

01739:     function installToolViewportGuard(panel) {
01740:         let pendingFrame = null;
01741: 
01742:         const requestClamp = () => {
01743:             if (pendingFrame != null) return;
01744:             pendingFrame = requestAnimationFrame(() => {
01745:                 pendingFrame = null;
01746:                 clampToolPanelToViewport(panel);
01747:             });
01748:         };
01749: 
01750:         const resetIosPosition = () => {
01751:             if (panel.classList.contains('mc-ios-safari')) {
01752:                 panel.style.left = '';
01753:                 panel.style.top = '';
01754:                 panel.style.right = '';
01755:                 panel.style.width = '';
01756:             }
01757:             requestClamp();
01758:         };
01759: 
01760:         window.addEventListener('resize', requestClamp, { passive: true });
01761:         window.addEventListener('orientationchange', resetIosPosition, { passive: true });
01762:         window.addEventListener('pageshow', requestClamp, { passive: true });
01763:         window.visualViewport?.addEventListener('resize', requestClamp, { passive: true });
01764:         window.visualViewport?.addEventListener('scroll', requestClamp, { passive: true });
01765: 
01766:         registerToolLifecycleCleanup(() => {
01767:             if (pendingFrame != null) {
01768:                 cancelAnimationFrame(pendingFrame);
01769:                 pendingFrame = null;
01770:             }
01771:             window.removeEventListener('resize', requestClamp);
01772:             window.removeEventListener('orientationchange', resetIosPosition);
01773:             window.removeEventListener('pageshow', requestClamp);
01774:             window.visualViewport?.removeEventListener('resize', requestClamp);
01775:             window.visualViewport?.removeEventListener('scroll', requestClamp);
01776:         });
01777: 
01778:         requestClamp();
01779:     }

01810: 
01811:             removeReadinessListeners();

01813:             return true;
01814:         }
01815: 
01816:         if (tryInitialise()) return;
01817: 
01818:         timer = setInterval(() => {
01819:             tries++;
01820:             if (tryInitialise()) return;
01821:             if (tries >= maximumTries) {
01822:                 clearInterval(timer);
01823:                 timer = null;
01824:             }
01825:         }, 500);
01826: 
01827:         registerToolLifecycleCleanup(
01828:             removeReadinessListeners
